cmake_minimum_required(VERSION 3.20)
project(cix_rocksdb_adapter LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CIX_NATIVE_INCLUDE_DIR "" CACHE PATH "installed CIX include directory")
set(CIX_NATIVE_LIBRARY "" CACHE FILEPATH "installed libcix_native library")
set(ROCKSDB_INCLUDE_DIR "" CACHE PATH "RocksDB v10.10.1 include directory")
set(ROCKSDB_LIBRARY "" CACHE FILEPATH "RocksDB v10.10.1 library")
foreach(value CIX_NATIVE_INCLUDE_DIR CIX_NATIVE_LIBRARY ROCKSDB_INCLUDE_DIR ROCKSDB_LIBRARY)
  if(NOT ${value})
    message(FATAL_ERROR "${value} is required; this adapter never fetches or falls back to a host SDK")
  endif()
endforeach()
include(CheckCXXSourceCompiles)
set(CMAKE_REQUIRED_INCLUDES ${ROCKSDB_INCLUDE_DIR})
check_cxx_source_compiles("#include <rocksdb/version.h>
#include <rocksdb/advanced_compression.h>
#include <rocksdb/compression_type.h>
#if ROCKSDB_MAJOR != 10 || ROCKSDB_MINOR != 10 || ROCKSDB_PATCH != 1
#error This adapter is pinned to RocksDB v10.10.1
#endif
int main() { return (int)rocksdb::kCustomCompression80; }" CIX_ROCKSDB_V10101_API)
if(NOT CIX_ROCKSDB_V10101_API)
  message(FATAL_ERROR "RocksDB v10.10.1 public CompressionManager headers are required")
endif()
add_library(cix_rocksdb_compression_manager STATIC cix_compression_manager.cc)
target_compile_features(cix_rocksdb_compression_manager PUBLIC cxx_std_20)
option(CIX_ROCKSDB_USE_RTTI "Match a RocksDB SDK built with USE_RTTI=ON" OFF)
if(CIX_ROCKSDB_USE_RTTI)
  target_compile_definitions(cix_rocksdb_compression_manager PUBLIC ROCKSDB_USE_RTTI)
elseif(MSVC)
  target_compile_options(cix_rocksdb_compression_manager PUBLIC /GR-)
else()
  target_compile_options(cix_rocksdb_compression_manager PUBLIC -fno-rtti)
endif()
target_include_directories(cix_rocksdb_compression_manager PUBLIC ${CMAKE_CURRENT_SOURCE_DIR} ${CIX_NATIVE_INCLUDE_DIR} ${ROCKSDB_INCLUDE_DIR})
target_link_libraries(cix_rocksdb_compression_manager PUBLIC ${CIX_NATIVE_LIBRARY} ${ROCKSDB_LIBRARY})
add_executable(cix_rocksdb_contract contract_test.cc)
target_link_libraries(cix_rocksdb_contract PRIVATE cix_rocksdb_compression_manager)
enable_testing()
add_test(NAME cix_rocksdb_contract COMMAND cix_rocksdb_contract)

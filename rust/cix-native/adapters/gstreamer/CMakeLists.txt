cmake_minimum_required(VERSION 3.16)
project(cix_gstreamer C)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GST REQUIRED IMPORTED_TARGET gstreamer-1.0 gstreamer-app-1.0)
set(CIX_INCLUDE_DIR "" CACHE PATH "Installed CIX SDK include directory")
set(CIX_NATIVE_LIBRARY "" CACHE FILEPATH "Installed CIX native library")
if(NOT CIX_INCLUDE_DIR OR NOT EXISTS "${CIX_INCLUDE_DIR}/cix_stream.h")
  message(FATAL_ERROR "Set CIX_INCLUDE_DIR to the selected installed CIX SDK include directory")
endif()
if(NOT CIX_NATIVE_LIBRARY OR NOT EXISTS "${CIX_NATIVE_LIBRARY}")
  message(FATAL_ERROR "Set CIX_NATIVE_LIBRARY to the selected installed CIX native library")
endif()
add_library(gstcixstream MODULE src/gstcixstream.c)
set_property(TARGET gstcixstream PROPERTY C_STANDARD 11)
target_include_directories(gstcixstream PRIVATE ${CIX_INCLUDE_DIR})
target_link_libraries(gstcixstream PRIVATE PkgConfig::GST ${CIX_NATIVE_LIBRARY})
target_compile_definitions(gstcixstream PRIVATE PACKAGE="cix" PACKAGE_NAME="cix")
# GStreamer discovers shared modules by the conventional libgst*.so filename.
set_target_properties(gstcixstream PROPERTIES OUTPUT_NAME "gstcixstream")
add_executable(cix-gstream-contract tests/contract.c)
set_property(TARGET cix-gstream-contract PROPERTY C_STANDARD 11)
target_link_libraries(cix-gstream-contract PRIVATE PkgConfig::GST)
enable_testing()
add_test(NAME cix-gstream-contract COMMAND cix-gstream-contract $<TARGET_FILE_DIR:gstcixstream>)

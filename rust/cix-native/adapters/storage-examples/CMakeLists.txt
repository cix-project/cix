cmake_minimum_required(VERSION 3.16)
project(cix_backup_example LANGUAGES CXX)
set(CIX_SDK_INCLUDE_DIR "" CACHE PATH "Installed CIX SDK include directory containing cix.h")
set(CIX_SDK_LIBRARY "" CACHE FILEPATH "Installed CIX SDK library")
if(NOT CIX_SDK_INCLUDE_DIR OR NOT EXISTS "${CIX_SDK_INCLUDE_DIR}/cix.h")
  message(FATAL_ERROR "Set CIX_SDK_INCLUDE_DIR to an installed SDK include directory containing cix.h")
endif()
if(NOT CIX_SDK_LIBRARY OR NOT EXISTS "${CIX_SDK_LIBRARY}")
  message(FATAL_ERROR "Set CIX_SDK_LIBRARY to the explicit installed CIX SDK library")
endif()
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
add_executable(cix-backup cix_backup.cpp)
target_compile_features(cix-backup PRIVATE cxx_std_17)
target_include_directories(cix-backup PRIVATE "${CIX_SDK_INCLUDE_DIR}")
target_link_libraries(cix-backup PRIVATE "${CIX_SDK_LIBRARY}" OpenSSL::Crypto)

cmake_minimum_required(VERSION 3.16)
project(cix_gio_adapter C)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GIO REQUIRED IMPORTED_TARGET gio-2.0)
find_path(CIX_INCLUDE_DIR cix_stream.h REQUIRED)
find_library(CIX_NATIVE_LIBRARY NAMES cix_native REQUIRED)
add_library(cix-gconverter SHARED cix-gconverter.c)
target_include_directories(cix-gconverter PRIVATE ${CIX_INCLUDE_DIR})
target_link_libraries(cix-gconverter PRIVATE PkgConfig::GIO ${CIX_NATIVE_LIBRARY})
add_executable(cix-gconverter-contract contract.c)
target_include_directories(cix-gconverter-contract PRIVATE ${CIX_INCLUDE_DIR})
target_link_libraries(cix-gconverter-contract PRIVATE cix-gconverter PkgConfig::GIO ${CIX_NATIVE_LIBRARY})
enable_testing()
add_test(NAME cix_gconverter_contract COMMAND cix-gconverter-contract)
